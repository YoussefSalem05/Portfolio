cmake_minimum_required(VERSION 3.16)
project(AmbulanceSystem LANGUAGES CXX)
add_executable(AmbulanceSystem Source.cpp Organizer.cpp Hospital.cpp Car.cpp Patients.cpp Output.cpp)
target_compile_features(AmbulanceSystem PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(AmbulanceSystem PRIVATE /W4 /permissive-)
else()
    target_compile_options(AmbulanceSystem PRIVATE -Wall -Wextra -Wpedantic)
endif()
set_property(TARGET AmbulanceSystem PROPERTY VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
configure_file(testload.txt testload.txt COPYONLY)
